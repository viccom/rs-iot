# rs-iot 前端实施计划（工业监控界面 + AI 浮动面板）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 Vue3 前端（工业监控界面 + AI 浮动面板），嵌入 gateway 提供，让用户经 AI 面板发指令后在同一界面看 rs-iot 实时数据验证结果。

**Architecture:** Vue3 + Vite + TS 工程放仓库根 `frontend/`，构建产物 `dist/` 经 rust-embed 嵌入 gateway（前端唯一入口）。监控界面走 rs-iot SSE + 轮询（经 gateway `/api/rsiot/*` 反代），AI 面板走 reasonix serve SSE（经 gateway `/api/agent/*` 反代）。两个 store 各管一条 EventSource。

**Tech Stack:** Vue3 + Vite + TypeScript + Pinia + vue-router；gateway 侧 rust-embed + axum 静态服务；测试 vitest（store 单测）+ axum 单测（gateway）。

**Spec:** `docs/superpowers/specs/2026-08-12-frontend-design.md`

```rust
//! crates/rsiot-gateway/src/static_assets.rs
//! 前端静态资源服务：rust-embed 嵌入 frontend/dist，SPA fallback 到 index.html。
//!
//! 路由策略（app.rs 注册顺序）：`/api/*` 走反代，其余走静态资源。
//! SPA fallback：未命中的非文件路径回退 index.html（vue-router history 模式）。

use axum::body::Body;
use axum::http::{header, HeaderValue, StatusCode, Uri};
use axum::response::{IntoResponse, Response};
use rust_embed::RustEmbed;

#[derive(RustEmbed)]
#[folder = "$CARGO_MANIFEST_DIR/../../frontend/dist"]
struct FrontendAsset;

/// 静态资源 handler：从路径取文件名，命中则返文件，未命中回退 index.html。
pub async fn serve(uri: Uri) -> Response {
    let path = uri.path().trim_start_matches('/');
    let file = if path.is_empty() { "index.html" } else { path };

    if let Some(content) = FrontendAsset::get(file) {
        let mime = mime_guess::from_path(file).first_or_octet_stream();
        Response::builder()
            .header(header::CONTENT_TYPE, HeaderValue::from_str(mime.as_ref()).unwrap())
            .body(Body::from(content.data.into_owned()))
            .unwrap()
    } else {
        // SPA fallback：非文件路径（如 /login）回退 index.html，交给 vue-router 处理。
        match FrontendAsset::get("index.html") {
            Some(content) => Response::builder()
                .header(header::CONTENT_TYPE, "text/html; charset=utf-8")
                .body(Body::from(content.data.into_owned()))
                .unwrap(),
            None => (StatusCode::NOT_FOUND, "frontend not built").into_response(),
        }
    }
}
```

- [ ] **Step 3: 改 auth.rs 支持 query token**

修改 `crates/rsiot-gateway/src/auth.rs` 的 `auth_middleware`，在 Authorization/Cookie 之后加 query 取 token：

```rust
// auth.rs auth_middleware 内，token 提取改为：
    let token = req
        .headers()
        .get(axum::http::header::AUTHORIZATION)
        .and_then(|v| v.to_str().ok())
        .map(|s| s.strip_prefix("Bearer ").unwrap_or(s).to_string())
        .or_else(|| {
            req.headers()
                .get(axum::http::header::COOKIE)
                .and_then(|v| v.to_str().ok())
                .and_then(|c| c.split("token=").nth(1).and_then(|s| s.split(';').next()))
                .map(|s| s.to_string())
        })
        .or_else(|| {
            // EventSource 不能设 header，支持 ?token=<jwt> query 参数。
            req.uri().query().and_then(|q| {
                q.split('&')
                    .find_map(|kv| kv.strip_prefix("token=").map(|s| s.to_string()))
            })
        });
```

// crates/rsiot-gateway/src/routes_gw.rs 加：
use axum::extract::State;
use axum::http::StatusCode;
use axum::Json;
use serde::{Deserialize, Serialize};
use std::sync::Arc;
use crate::proxy::GwState;

#[derive(Deserialize)]
pub struct LoginReq {
    pub user_id: String,
    pub pass: String,
}

/// POST /api/gw/login：转发 rs-iot /api/v1/user/login，透传返的 JWT。
///
/// gateway 不验密码（rs-iot 验 bcrypt），纯转发。jwt_secret 与 rs-iot 同值，
/// 故 rs-iot 颁的 JWT gateway 也认。
pub async fn login(State(st): State<Arc<GwState>>, Json(req): Json<LoginReq>) -> Response {
    let url = format!("{}/api/v1/user/login", st.rsiot_base);
    let resp = match st.client.post(&url).json(&req).send().await {
        Ok(r) => r,
        Err(_) => return (StatusCode::BAD_GATEWAY, "rs-iot unreachable").into_response(),
    };
    let status = resp.status();
    let text = resp.text().await.unwrap_or_default();
    // 透传 rs-iot 响应（含 {token, user} 或错误）。
    Response::builder()
        .status(status)
        .header("content-type", "application/json")
        .body(Body::from(text))
        .unwrap()
}
```

并在 routes_gw.rs 顶部补 `use axum::response::{IntoResponse, Response}; use axum::body::Body;`。

// frontend/src/stores/auth.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'

export const useAuthStore = defineStore('auth', () => {
  const token = ref(localStorage.getItem('rsiot_token') || '')
  const user = ref(localStorage.getItem('rsiot_user') || '')

  function setAuth(t: string, u: string) {
    token.value = t
    user.value = u
    localStorage.setItem('rsiot_token', t)
    localStorage.setItem('rsiot_user', u)
  }

  function clear() {
    token.value = ''
    user.value = ''
    localStorage.removeItem('rsiot_token')
    localStorage.removeItem('rsiot_user')
  }

  return { token, user, setAuth, clear }
})
```

```typescript
// frontend/src/api/client.ts
import { useAuthStore } from '../stores/auth'
import { router } from '../router'

/** fetch 封装：自动带 JWT，401 清 token 跳登录。 */
export async function api(path: string, opts: RequestInit = {}) {
  const headers = new Headers(opts.headers)
  if (useAuthStore().token) headers.set('Authorization', `Bearer ${useAuthStore().token}`)
  if (!headers.has('Content-Type') && opts.body) headers.set('Content-Type', 'application/json')
  const r = await fetch(path, { ...opts, headers })
  if (r.status === 401) {
    useAuthStore().clear()
    router.push('/login')
    throw new Error('unauthorized')
  }
  return r
}
```

```typescript
// frontend/src/stores/monitor.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import type { InstanceStat, LogEntry, PacketEntry, StateEvent, RtdataCell } from '../types'
import { useAuthStore } from './auth'

const MAX_LOGS = 500
const MAX_PACKETS = 200

export const useMonitorStore = defineStore('monitor', () => {
  const instances = ref<InstanceStat[]>([])
  const logs = ref<LogEntry[]>([])
  const packets = ref<PacketEntry[]>([])
  const rtdata = ref<Record<string, Record<string, RtdataCell>>>({})
  const sysstatus = ref<Record<string, unknown>>({})
  const selectedInstance = ref<string>('')

  let sse: EventSource | null = null
  let rtdataTimer: ReturnType<typeof setInterval> | null = null
  let sysstatusTimer: ReturnType<typeof setInterval> | null = null

  function onLog(e: LogEntry) {
    logs.value.unshift(e)
    if (logs.value.length > MAX_LOGS) logs.value.length = MAX_LOGS
  }
  function onPacket(e: PacketEntry) {
    packets.value.unshift(e)
    if (packets.value.length > MAX_PACKETS) packets.value.length = MAX_PACKETS
  }
  function onState(e: StateEvent) {
    const i = instances.value.findIndex(x => x.inst_id === e.inst_id)
    if (i >= 0) instances.value[i] = { ...instances.value[i], state: e.state }
    else instances.value.push({ inst_id: e.inst_id, app_code: e.app_code, host_kind: e.host_kind, state: e.state, health: null })
  }

  async function loadInstances() {
    const r = await fetch('/api/rsiot/inststatus', { headers: { Authorization: `Bearer ${useAuthStore().token}` } })
    if (r.ok) instances.value = Object.values(await r.json()) as InstanceStat[]
  }
  async function loadRtdata() {
    const r = await fetch('/api/rsiot/rtdatas', { headers: { Authorization: `Bearer ${useAuthStore().token}` } })
    if (r.ok) rtdata.value = await r.json()
  }
  async function loadSysstatus() {
    const r = await fetch('/api/rsiot/sysstatus', { headers: { Authorization: `Bearer ${useAuthStore().token}` } })
    if (r.ok) sysstatus.value = await r.json()
  }

  /** 启动 SSE + 轮询。登录后调用。 */
  function start() {
    // SSE：rs-iot 有 event: 头，用 addEventListener(kind) 分发。token 走 query（EventSource 不能设 header）。
    const tok = useAuthStore().token
    sse = new EventSource(`/api/rsiot/events?types=log,packet,state&token=${encodeURIComponent(tok)}`)
    sse.addEventListener('log', e => onLog(JSON.parse((e as MessageEvent).data)))
    sse.addEventListener('packet', e => onPacket(JSON.parse((e as MessageEvent).data)))
    sse.addEventListener('state', e => onState(JSON.parse((e as MessageEvent).data)))
    sse.onerror = () => { /* 浏览器自动重连；断线兜底 loadLogs（M4 完善） */ }
    loadInstances(); loadRtdata(); loadSysstatus()
    rtdataTimer = setInterval(loadRtdata, 2000)
    sysstatusTimer = setInterval(loadSysstatus, 5000)
    document.addEventListener('visibilitychange', onVisibility)
  }

  function onVisibility() {
    if (document.hidden) { if (rtdataTimer) clearInterval(rtdataTimer); if (sysstatusTimer) clearInterval(sysstatusTimer) }
    else { rtdataTimer = setInterval(loadRtdata, 2000); sysstatusTimer = setInterval(loadSysstatus, 5000) }
  }

  function stop() {
    sse?.close(); sse = null
    if (rtdataTimer) clearInterval(rtdataTimer)
    if (sysstatusTimer) clearInterval(sysstatusTimer)
    document.removeEventListener('visibilitychange', onVisibility)
  }

  function selectInstance(id: string) { selectedInstance.value = id }

  return { instances, logs, packets, rtdata, sysstatus, selectedInstance, onLog, onPacket, onState, loadInstances, start, stop, selectInstance }
})
```

```typescript
// frontend/src/stores/agent.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import type { ChatMessage, ServeEvent } from '../types'
import { useAuthStore } from './auth'
import { useMonitorStore } from './monitor'
import { api } from '../api/client'

export const useAgentStore = defineStore('agent', () => {
  const messages = ref<ChatMessage[]>([])
  const running = ref(false)
  const approval = ref<{ id: string; tool: string; args?: unknown } | null>(null)
  let nextId = 1
  let sse: EventSource | null = null

  /** 处理 serve SSE 事件（无 event: 头，kind 在 JSON body）。 */
  function onEvent(e: ServeEvent) {
    const cur = () => messages.value.at(-1)
    switch (e.kind) {
      case 'turn_started':
        messages.value.push({ id: nextId++, role: 'assistant', text: '', toolCalls: [] })
        running.value = true
        break
      case 'text':
        if (cur()?.role === 'assistant') cur()!.text += e.text || ''
        break
      case 'reasoning':
        if (cur()?.role === 'assistant') cur()!.reasoning = (cur()!.reasoning || '') + (e.text || '')
        break
      case 'tool_dispatch':
        if (e.tool) cur()?.toolCalls?.push({ id: e.tool.id, name: e.tool.name })
        break
      case 'tool_result': {
        const tc = cur()?.toolCalls?.find(t => t.id === (e.tool as { id?: string })?.id)
        if (tc) { tc.result = e; tc.error = false }
        break
      }
      case 'approval_request':
        approval.value = { id: (e.approval_request as { id: string }).id, tool: (e.approval_request as { tool: string }).tool, args: (e.approval_request as { args?: unknown }).args }
        break
      case 'turn_done':
        running.value = false
        break
    }
  }

  /** 发任务。附当前监控选中实例上下文（页面感知）。 */
  async function submit(input: string) {
    messages.value.push({ id: nextId++, role: 'user', text: input })
    const monitor = useMonitorStore()
    const ctx = monitor.selectedInstance ? `\n[上下文: 当前查看实例 ${monitor.selectedInstance}]` : ''
    await api('/api/agent/submit', { method: 'POST', body: JSON.stringify({ input: input + ctx }) })
  }

  async function approve(decision: boolean) {
    if (!approval.value) return
    await api('/api/agent/approve', { method: 'POST', body: JSON.stringify({ approval_id: approval.value.id, decision }) })
    approval.value = null
  }

  async function cancel() {
    await api('/api/agent/cancel', { method: 'POST', body: '{}' })
  }

  function start() {
    const tok = useAuthStore().token
    sse = new EventSource(`/api/agent/events?token=${encodeURIComponent(tok)}`)
    sse.onmessage = ev => { try { onEvent(JSON.parse(ev.data)) } catch { /* ignore */ } }
    sse.onerror = () => { /* 浏览器自动重连 */ }
  }

  function stop() { sse?.close(); sse = null }

  return { messages, running, approval, onEvent, submit, approve, cancel, start, stop }
})
```

async function loadLogsRecent() {
  const r = await fetch('/api/rsiot/logs/recent?limit=200', { headers: { Authorization: `Bearer ${useAuthStore().token}` } })
  if (r.ok) {
    const entries: LogEntry[] = await r.json()
    // 合并：去重（按 ts+message），保最新 500
    const seen = new Set(logs.value.map(l => `${l.ts}|${l.message}`))
    for (const e of entries.reverse()) {
      const k = `${e.ts}|${e.message}`; if (!seen.has(k)) { logs.value.unshift(e); seen.add(k) }
    }
    if (logs.value.length > 500) logs.value.length = 500
    offline.value = false
  }
}
