# 50万点/秒采集入库改造 · 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 rs-iot 单进程入库吞吐从 ~10-20万点/秒提升到稳定 50万点/秒,支撑电网区域中心 100 实例 × 5000 点/秒。

**Architecture:** pipeline 单消费者 → dispatcher + N worker(按 devid hash 分片);`rtdata::write` 逐点 HSET → `write_batch` 用 lux `PreparedPipeline` + `execute_prepared_pipeline_discard` 批量提交(HSET 多对);采集 ticker 加 jitter 错峰。不动 Source trait / Sample 结构 / lux / hub 上游契约。

**Tech Stack:** Rust + tokio + lux(`EmbeddedClient::clone`/`PreparedPipeline`/`execute_prepared_pipeline_discard`)。零新依赖(jitter hash 用手写 FNV-1a)。

**Spec:** `docs/superpowers/specs/2026-08-11-500k-throughput-design.md`

```rust
// examples/bench_rtdata.rs
//! rtdata 写入吞吐 bench:A(逐点HSET) / B(HSET多对) / C(Pipeline) / D(多clone并发)。
//! 运行:cargo run --release --example bench_rtdata
use std::time::Instant;

#[tokio::main(flavor = "multi_thread", worker_threads = 4)]
async fn main() {
    let dir = std::env::temp_dir().join(format!("bench-lux-{}", std::process::id()));
    let _ = std::fs::remove_dir_all(&dir);
    std::fs::create_dir_all(&dir).unwrap();
    let mut sc = lux::ServerConfig::default();
    sc.data_dir = dir.to_str().unwrap().to_string();
    sc.storage = lux::StorageConfig { mode: lux::StorageMode::Tiered, dir: sc.data_dir.clone() };
    sc.save_interval = std::time::Duration::from_secs(60);
    let handle = lux::run_with_config(sc).await.unwrap();
    let client = handle.client();

    let n_points = 5_000;       // 单 Sample 点数
    let n_samples = 100;        // 一轮 Sample 数(模拟 100 实例)
    let rounds = 5;             // 重复轮次取平均

    // 场景 A:逐点 HSET(现状,baseline)
    let t = bench_a(&client, n_points, n_samples, rounds).await;
    println!("A 逐点HSET:           {t:>10.0} 点/秒");

    // 场景 B:HSET 多对(收益①)
    let t = bench_b(&client, n_points, n_samples, rounds).await;
    println!("B HSET多对:           {t:>10.0} 点/秒");

    // 场景 C:PreparedPipeline discard(收益②)
    let t = bench_c(&client, n_points, n_samples, rounds).await;
    println!("C Pipeline discard:   {t:>10.0} 点/秒");

    // 场景 D:N×clone 并发(收益③,N=4)
    let t = bench_d(&client, n_points, n_samples, rounds, 4).await;
    println!("D 4×clone 并发:       {t:>10.0} 点/秒");

    client.execute("SAVE", &[]).await.ok();
    handle.shutdown_and_wait().await.unwrap();
    let _ = std::fs::remove_dir_all(&dir);
}

fn mk_samples(n_points: usize, n_samples: usize) -> Vec<rs_iot::types::Sample> {
    // 构造 n_samples 个 Sample,各带 n_points 点(devid 互异)
    (0..n_samples).map(|s| rs_iot::types::Sample {
        hwid: rs_iot::types::Hwid::from("bench".into()),
        devid: format!("d{s}"),
        values: (0..n_points).map(|i| (format!("t{i}"), rs_iot::types::SampleValue::Float(i as f64))).collect(),
        timestamp: 0,
    }).collect()
}

async fn bench_a(c: &lux::EmbeddedClient, np: usize, ns: usize, rounds: usize) -> f64 {
    // 复刻 rtdata::write 逐点逻辑
    let mut total_pts = 0u64; let mut elapsed = std::time::Duration::ZERO;
    for _ in 0..rounds {
        let samples = mk_samples(np, ns);
        let t0 = Instant::now();
        for s in &samples {
            let key = format!("rtdata:{}", s.devid);
            for (tag, v) in &s.values {
                let cell = format!("[0,\"\",{}]", match v { rs_iot::types::SampleValue::Float(f)=>f.to_string(), _=>String::new() });
                c.execute("HSET", &[&key, tag, &cell]).await.unwrap();
            }
        }
        elapsed += t0.elapsed(); total_pts += (np * ns) as u64;
    }
    total_pts as f64 / elapsed.as_secs_f64()
}

async fn bench_b(c: &lux::EmbeddedClient, np: usize, ns: usize, rounds: usize) -> f64 {
    // HSET 多对:每 Sample 一次 execute
    let mut total_pts = 0u64; let mut elapsed = std::time::Duration::ZERO;
    for _ in 0..rounds {
        let samples = mk_samples(np, ns);
        let t0 = Instant::now();
        for s in &samples {
            let key = format!("rtdata:{}", s.devid);
            let mut argv: Vec<Vec<u8>> = vec![b"HSET".to_vec(), key.into_bytes()];
            for (tag, v) in &s.values {
                argv.push(tag.as_bytes().to_vec());
                argv.push(format!("[0,\"\",{}]", match v { rs_iot::types::SampleValue::Float(f)=>f.to_string(), _=>String::new() }).into_bytes());
            }
            c.execute_argv(argv).await.unwrap();
        }
        elapsed += t0.elapsed(); total_pts += (np * ns) as u64;
    }
    total_pts as f64 / elapsed.as_secs_f64()
}

async fn bench_c(c: &lux::EmbeddedClient, np: usize, ns: usize, rounds: usize) -> f64 {
    // PreparedPipeline:全批一次提交
    let mut total_pts = 0u64; let mut elapsed = std::time::Duration::ZERO;
    for _ in 0..rounds {
        let samples = mk_samples(np, ns);
        let t0 = Instant::now();
        let mut pipe = lux::PreparedPipeline::new();
        for s in &samples {
            let key = format!("rtdata:{}", s.devid);
            let mut argv: Vec<String> = vec!["HSET".into(), key];
            for (tag, v) in &s.values {
                argv.push(tag.clone());
                argv.push(format!("[0,\"\",{}]", match v { rs_iot::types::SampleValue::Float(f)=>f.to_string(), _=>String::new() }));
            }
            pipe.push_argv(argv).unwrap();
        }
        c.execute_prepared_pipeline_discard(&pipe).await.unwrap();
        elapsed += t0.elapsed(); total_pts += (np * ns) as u64;
    }
    total_pts as f64 / elapsed.as_secs_f64()
}

async fn bench_d(c: &lux::EmbeddedClient, np: usize, ns: usize, rounds: usize, workers: usize) -> f64 {
    // N 个 clone 各处理 ns/workers Sample 并发
    use std::sync::Arc;
    let mut total_pts = 0u64; let mut elapsed = std::time::Duration::ZERO;
    for _ in 0..rounds {
        let samples = mk_samples(np, ns);
        let t0 = Instant::now();
        let chunks: Vec<Vec<rs_iot::types::Sample>> = samples.into_iter().collect::<Vec<_>>()
            .chunks((ns + workers - 1) / workers.max(1)).map(|c| c.to_vec()).collect();
        let mut handles = Vec::new();
        for (i, chunk) in chunks.into_iter().enumerate() {
            let ci = c.clone();  // 独立 session Mutex
            handles.push(tokio::spawn(async move {
                let mut pipe = lux::PreparedPipeline::new();
                for s in &chunk {
                    let key = format!("rtdata:{}", s.devid);
                    let mut argv: Vec<String> = vec!["HSET".into(), key];
                    for (tag, v) in &s.values { argv.push(tag.clone()); argv.push(format!("[0,\"\",{}]", match v { rs_iot::types::SampleValue::Float(f)=>f.to_string(), _=>String::new() })); }
                    pipe.push_argv(argv).unwrap();
                }
                ci.execute_prepared_pipeline_discard(&pipe).await.unwrap();
                let _ = i;
            }));
        }
        for h in handles { h.await.unwrap(); }
        elapsed += t0.elapsed(); total_pts += (np * ns) as u64;
    }
    total_pts as f64 / elapsed.as_secs_f64()
}
```

在 `src/db/rtdata.rs` 的现有 `write` 函数后加:
```rust
/// 批量写入多个 Sample 到 lux rtdata(收益①②:同 devid 多点合成一条 HSET 多对,
/// 跨 devid 用 PreparedPipeline 一次提交,lux 同 shard 自动合并 WAL)。
///
/// 同 devid 跨 Sample → 后入覆盖(BTreeMap insert 语义,HSET 覆盖一致)。
/// 空批直接返回(不提交)。local_time 批内取一次。
pub async fn write_batch(c: &EmbeddedClient, samples: &[crate::types::Sample]) -> Result<(), RsIotError> {
    if samples.is_empty() {
        return Ok(());
    }
    use std::collections::BTreeMap;
    let local_time = chrono::Local::now().format("%Y-%m-%d %H:%M:%S").to_string();
    // devid -> Vec<(field, cell)>;Vec 保序(同 devid 多 Sample 后入覆盖靠 field 名相同,HSET 多对后值覆盖前值)
    let mut by_devid: BTreeMap<&str, Vec<(String, String)>> = BTreeMap::new();
    for s in samples {
        let entry = by_devid.entry(s.devid.as_str()).or_default();
        for (tag, val) in &s.values {
            let field = if tag.contains(s.devid.as_str()) {
                tag.clone()
            } else {
                format!("{}.{}", s.devid, tag)
            };
            let val_str = match val {
                crate::types::SampleValue::Float(f) => f.to_string(),
                crate::types::SampleValue::Int(i) => i.to_string(),
                crate::types::SampleValue::Bool(b) => b.to_string(),
                crate::types::SampleValue::Str(st) => st.clone(),
            };
            let cell = serde_json::to_string(&serde_json::json!([s.timestamp, local_time, val_str]))?;
            entry.push((field, cell));
        }
    }
    let mut pipe = lux::PreparedPipeline::new();
    for (devid, pairs) in &by_devid {
        let key = format!("rtdata:{devid}");
        let mut argv: Vec<String> = Vec::with_capacity(2 + pairs.len() * 2);
        argv.push("HSET".to_string());
        argv.push(key);
        for (f, v) in pairs {
            argv.push(f.clone());
            argv.push(v.clone());
        }
        pipe.push_argv(argv)
            .map_err(|e| RsIotError::Lux(lux::LuxError::Other(format!("push_argv: {e}"))))?;
    }
    c.execute_prepared_pipeline_discard(&pipe).await?;
    Ok(())
}
```
**注:** 若 `lux::LuxError` 无 `Other` 变体,改用现有变体(以 lux 实际为准,`rg "pub enum LuxError" ../lux/src`)。

```rust
//! 采集 ticker jitter helper:FNV-1a(inst_id) % period_ms 确定性相位,
//! 让多实例 ticker 首次 tick 散布开,避免整点同步触发脉冲。

use std::time::Duration;

/// FNV-1a 64bit(确定性,无外部依赖)。
fn fnv1a_64(s: &str) -> u64 {
    let mut h: u64 = 0xcbf29ce484222325;
    for b in s.bytes() {
        h ^= b as u64;
        h = h.wrapping_mul(0x100000001b3);
    }
    h
}

/// 返回带 jitter 相位的 interval:首次 tick 延迟 `fnv1a(inst_id) % period`。
/// 同 inst_id + 同 period → 完全相同(确定性);不同 inst_id → 散布在 [0, period)。
pub fn phase_ms(inst_id: &str, period: Duration) -> u64 {
    let period_ms = period.as_millis().max(1) as u64;
    fnv1a_64(inst_id) % period_ms
}

#[cfg(test)]
mod tests {
    use super::*;
    #[test]
    fn phase_deterministic_same_input() {
        let p = Duration::from_secs(1);
        assert_eq!(phase_ms("instA", p), phase_ms("instA", p));
    }
    #[test]
    fn phase_spreads_across_instances() {
        let p = Duration::from_secs(1);
        let phases: Vec<u64> = ["i0","i1","i2","i3","i4","i5","i6","i7"].iter().map(|s| phase_ms(s, p)).collect();
        // 8 个实例,相位应散布(至少 4 个不同值,非全相同)
        let unique: std::collections::HashSet<u64> = phases.iter().copied().collect();
        assert!(unique.len() >= 4, "相位应散布,实际 unique={unique:?}");
        // 全部在 [0, 1000)
        assert!(phases.iter().all(|&x| x < 1000));
    }
}
```

```rust
//! 数据管道:dispatcher 按 devid hash 把 Sample 路由到 N 个 worker,
//! 每 worker 持独立 EmbeddedClient::clone,攒批窗口 50ms → rtdata::write_batch → lux。
//!
//! worker 是消费者侧 fan-out,与 Source 解耦(单 Source 启停不影响)。
//! lux 不动、Source trait / Sample 结构不动。

use crate::db;
use crate::host::HostContext;
use crate::plugin::traits::Processor;
use crate::types::Sample;
use std::sync::Arc;
use std::time::Duration;
use tokio::sync::mpsc;
use tokio_util::sync::CancellationToken;

/// devid → worker idx(FNV-1a,与 ticker helper 同算法保一致)。
fn route(devid: &str, n: usize) -> usize {
    let mut h: u64 = 0xcbf29ce484222325;
    for b in devid.bytes() { h ^= b as u64; h = h.wrapping_mul(0x100000001b3); }
    (h as usize) % n.max(1)
}

fn auto_workers(requested: usize) -> usize {
    if requested > 0 { return requested; }
    let cpu = std::thread::available_parallelism().map(|n| n.get()).unwrap_or(4);
    cpu.min(8)
}

/// 启动 dispatcher + N worker。dispatcher 持有 hub_rx,按 devid 路由。
pub async fn run(
    ctx: HostContext,
    mut hub_rx: mpsc::Receiver<Sample>,
    processor: Arc<dyn Processor>,
    workers: usize,
    token: CancellationToken,
) {
    let n = auto_workers(workers);
    tracing::info!("[pipeline] starting {n} worker(s)");
    ctx.metrics.set_workers_active(n);

    let (worker_txs, worker_tasks): (Vec<mpsc::Sender<Sample>>, Vec<_>) = (0..n)
        .map(|_| {
            let (wtx, wrx) = mpsc::channel::<Sample>(256);
            let ctx_w = ctx.clone();
            let proc_w = processor.clone();
            let lux_w = ctx.lux.clone();       // clone:独立 session Mutex,共享 store
            let token_w = token.child_token();
            let task = tokio::spawn(worker_run(ctx_w, wrx, proc_w, lux_w, token_w));
            (wtx, task)
        })
        .unzip();

    // dispatcher:唯一 hub_rx 消费者,路由 + 背压 send。
    loop {
        tokio::select! {
            _ = token.cancelled() => { tracing::info!("[pipeline] dispatcher cancelled"); break; }
            sample = hub_rx.recv() => {
                let s = match sample { Some(s) => s, None => { tracing::info!("[pipeline] hub closed"); break; } };
                ctx.metrics.hub_depth(hub_rx.len());
                let idx = route(&s.devid, n);
                if worker_txs[idx].send(s).await.is_err() {
                    tracing::error!("[pipeline] worker {idx} closed (devid 分片停入库)");
                }
            }
        }
    }
    // 退出:cancel 所有 worker token(经父 token 已级联)+ 等 task
    for (i, t) in worker_tasks.into_iter().enumerate() {
        let _ = tokio::time::timeout(Duration::from_secs(3), t).await;
        tracing::info!("[pipeline] worker {i} joined");
    }
    tracing::info!("[pipeline] stopped");
}

/// 单 worker:攒批窗口 50ms → write_batch。所有 lux 错误 catch 降级,不 propagate。
async fn worker_run(
    ctx: HostContext,
    mut rx: mpsc::Receiver<Sample>,
    processor: Arc<dyn Processor>,
    lux: lux::EmbeddedClient,
    token: CancellationToken,
) {
    let mut buf: Vec<Sample> = Vec::new();
    let mut flush = tokio::time::interval(Duration::from_millis(50));
    flush.tick().await; // 跳过首个立即 tick
    loop {
        tokio::select! {
            _ = token.cancelled() => break,
            s = rx.recv() => match s {
                Some(s) => { if let Some(p) = processor.process(s).await { buf.push(p); } }
                None => break,
            },
            _ = flush.tick() => {
                if buf.is_empty() { continue; }
                let n = buf.len();
                match db::rtdata::write_batch(&lux, &buf).await {
                    Ok(()) => { ctx.metrics.pipeline_processed_batch(n); ctx.metrics.pipeline_batch(); }
                    Err(e) => { ctx.metrics.pipeline_write_fail(); tracing::warn!("[pipeline] write_batch failed: {e}"); }
                }
                buf.clear();
            }
        }
    }
    ctx.metrics.worker_exited();
    tracing::info!("[pipeline] worker exited");
}
```
